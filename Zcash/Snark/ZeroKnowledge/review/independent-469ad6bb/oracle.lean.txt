def retryAttempt : ProverAttemptResult := ⟨[1], [], .failed .retryRandomness⟩

def completeAttempt : ProverAttemptResult := ⟨[3], [], .complete⟩
def countStep (state : Nat) (attempt : Option ProverAttemptResult) := (attempt, state + 1)

    ({true} : Set Bool) (fun _ => ()) () ∈ retryStreamNontermination := by
  intro index
  induction index with
